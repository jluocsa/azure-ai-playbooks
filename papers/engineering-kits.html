Why It Matters

Writing static AST linters and normalized PTU KQL queries from scratch slows down migration readiness work.

Key Takeaway

Zero-dependency pcr_lint.py scans Python/TS/JS/C# repos in seconds, and ptu-cache-release-gate.kql computes two-regime PTU & shard SLOs.

Practical Application

Download or 1-click copy either file below directly into target repositories and Log Analytics workspaces (MIT licensed).

Both artifacts below are rendered directly from the live kit files in this repository so you can inspect, search, 1-click copy, or download the raw CLI/query file into customer repositories.

1. PCR001–PCR006 Static CI Linter (pcr_lint.py) ​

Zero-dependency Python 3 AST + regex scanner checking Python, TypeScript, JavaScript, and C# codebases for:

  • PCR001: Missing prompt_cache_key or high-cardinality user_id / request_id / uuid4 / engagement_id / matter_id / client_id / cost_center key interpolation on responses.create / completions.create.
  • PCR002: Live timestamps, UUIDs, or per-request context (req_ctx) injected near leading system / developer prompts.
  • PCR003: json.dumps(...) on tool, schema, context, card, rubric, or taxonomy objects without sort_keys=True.
  • PCR004: Dynamic rag_chunks / search_results placed ahead of static policy or domain methodology.
  • PCR005: Deprecated prompt_cache_retention on GPT-5.6+ or prompt_cache_breakpoint targeting GPT-5.6 PTU-M.
  • PCR006: Wrapper modules reading prompt_tokens or usage.input_tokens without extracting cached_tokens.

Run in Local Dev or Pull-Request CI ​

bash
python3 pcr_lint.py ./src
python3 pcr_lint.py ./src --json

python
#!/usr/bin/env python3
"""PCR001-PCR006 Static Prompt-Caching Readiness Linter (Python AST + Regex).

Zero-dependency CLI scanner for GPT-5.6, GPT-6, and GPT-6.1 migrations.
Last verified: 2026-11-18 21:21 PST
Usage:
    python pcr_lint.py ./src
    python pcr_lint.py ./src --json

License: MIT
"""

from __future__ import annotations

import argparse
import ast
import json
import re
import sys
from dataclasses import asdict, dataclass
from pathlib import Path

SKIP_DIRS = {".git", ".venv", "venv", "node_modules", "__pycache__", "dist", "build"}
TARGET_EXTS = {".py", ".ts", ".js", ".cs"}

VOLATILE_TOKENS_RE = re.compile(
    r"\b(datetime\.now|datetime\.utcnow|DateTime\.UtcNow|Date\.now|time\.time|uuid4|request_timestamp|req_ctx|request_ctx)\b"
)
DYNAMIC_KEY_RE = re.compile(
    r"(user_id|request_id|uuid4|trace_id|engagement_id|engagement_code|matter_id|client_id|cost_center)",
    re.IGNORECASE,
)
GPT56_OR_6_RE = re.compile(r"gpt-(5\.6|6(\.1)?)", re.IGNORECASE)
PTU_56_RE = re.compile(r"gpt-5[.-]6.*ptu|ptu.*gpt-5[.-]6", re.IGNORECASE)


@dataclass
class Finding:
    rule: str
    file: str
    line: int
    message: str


def scan_python_ast(path: Path, source: str) -> list[Finding]:
    findings: list[Finding] = []
    try:
        tree = ast.parse(source, filename=str(path))
    except SyntaxError:
        return findings

    rel = path.as_posix()
    for node in ast.walk(tree):
        if not isinstance(node, ast.Call):
            continue

        func_src = ast.unparse(node.func) if hasattr(ast, "unparse") else ""
        call_src = ast.unparse(node) if hasattr(ast, "unparse") else ""
        kw_names = {kw.arg: kw.value for kw in node.keywords if kw.arg}

        # PCR001: Missing or per-user/request prompt_cache_key on responses/chat.completions create
        if func_src.endswith(("responses.create", "completions.create")):
            if "prompt_cache_key" not in kw_names:
                findings.append(
                    Finding(
                        "PCR001",
                        rel,
                        node.lineno,
                        "LLM create() call omits prompt_cache_key (required for deterministic GPT-5.6+ routing).",
                    )
                )
            else:
                key_expr = ast.unparse(kw_names["prompt_cache_key"]) if hasattr(ast, "unparse") else ""
                if DYNAMIC_KEY_RE.search(key_expr):
                    findings.append(
                        Finding(
                            "PCR001",
                            rel,
                            node.lineno,
                            f"prompt_cache_key interpolates high-cardinality identifier ({key_expr}); shard by hash(thread_id) % N instead.",
                        )
                    )

            # PCR005: Deprecated prompt_cache_retention on GPT-5.6+
            if "prompt_cache_retention" in kw_names and GPT56_OR_6_RE.search(call_src):
                findings.append(
                    Finding(
                        "PCR005",
                        rel,
                        node.lineno,
                        "prompt_cache_retention is deprecated on GPT-5.6+; use prompt_cache_options={'ttl': '30m'|'12h'}.",
                    )
                )

        # PCR003: json.dumps on schema / card / tool / context objects without sort_keys=True
        if func_src == "json.dumps" and node.args:
            arg0 = ast.unparse(node.args[0]) if hasattr(ast, "unparse") else ""
            if re.search(r"(tool|schema|card|context|ctx|rubric|taxonomy)", arg0, re.IGNORECASE):
                sort_kw = kw_names.get("sort_keys")
                is_sorted = isinstance(sort_kw, ast.Constant) and sort_kw.value is True
                if not is_sorted:
                    findings.append(
                        Finding(
                            "PCR003",
                            rel,
                            node.lineno,
                            f"json.dumps({arg0}) lacks sort_keys=True; key-order drift breaks Pos 1-4 prefix caching.",
                        )
                    )

    return findings


def scan_text_patterns(path: Path, source: str) -> list[Finding]:
    findings: list[Finding] = []
    rel = path.as_posix()
    lines = source.splitlines()

    for idx, line in enumerate(lines, start=1):
        # PCR002: Volatile timestamp/UUID injected inside system/developer message definition
        if re.search(r'role["\']?\s*[:=]\s*["\'](system|developer)["\']', line) or "sys_prompt" in line or "sys_text" in line:
            window = "\n".join(lines[max(0, idx - 3) : min(len(lines), idx + 4)])
            if VOLATILE_TOKENS_RE.search(window):
                findings.append(
                    Finding(
                        "PCR002",
                        rel,
                        idx,
                        "Live timestamp/UUID detected near system/developer prompt; keep Pos 1-3 100% static.",
                    )
                )

        # PCR004: RAG chunks placed ahead of static instructions in same template
        if re.search(r"(rag_chunks|search_results).*?(static_policy|system_prompt|domain_methodology)", line):
            findings.append(
                Finding(
                    "PCR004",
                    rel,
                    idx,
                    "Dynamic RAG/search content appears ahead of static instructions; move volatile grounding to Pos 6.",
                )
            )

        # PCR005: Explicit breakpoints sent to GPT-5.6 PTU-M deployment
        if "prompt_cache_breakpoint" in line and PTU_56_RE.search(source):
            findings.append(
                Finding(
                    "PCR005",
                    rel,
                    idx,
                    "prompt_cache_breakpoint detected alongside GPT-5.6 PTU identifier (breakpoints unsupported on 5.6 PTU-M).",
                )
            )

    # PCR006: Logs prompt_tokens or usage.input_tokens without reading cached_tokens anywhere in the wrapper module
    has_usage_tokens = "prompt_tokens" in source or bool(re.search(r"\busage\.input_tokens\b", source))
    if has_usage_tokens and "cached_tokens" not in source:
        line_no = next(
            (i for i, l in enumerate(lines, 1) if "prompt_tokens" in l or "input_tokens" in l),
            1,
        )
        findings.append(
            Finding(
                "PCR006",
                rel,
                line_no,
                "Module reads usage token counts without extracting input_tokens_details/prompt_tokens_details.cached_tokens.",
            )
        )

    return findings


def scan_path(target: Path) -> list[Finding]:
    files = [target] if target.is_file() else [
        p for p in target.rglob("*") if p.is_file() and p.suffix.lower() in TARGET_EXTS and not (set(p.parts) & SKIP_DIRS)
    ]
    all_findings: list[Finding] = []
    for f in sorted(files):
        text = f.read_text(encoding="utf-8", errors="ignore")
        if f.suffix.lower() == ".py":
            all_findings.extend(scan_python_ast(f, text))
        all_findings.extend(scan_text_patterns(f, text))
    # Deduplicate by (rule, file, line)
    seen: set[tuple[str, str, int]] = set()
    unique: list[Finding] = []
    for item in all_findings:
        key = (item.rule, item.file, item.line)
        if key not in seen:
            seen.add(key)
            unique.append(item)
    return unique


def main() -> int:
    parser = argparse.ArgumentParser(description="PCR001-PCR006 static prompt-caching readiness linter.")
    parser.add_argument("paths", nargs="+", help="Files or directories to scan")
    parser.add_argument("--json", action="store_true", help="Emit findings as JSON")
    args = parser.parse_args()

    findings: list[Finding] = []
    for raw in args.paths:
        findings.extend(scan_path(Path(raw)))

    if args.json:
        print(json.dumps([asdict(f) for f in findings], indent=2))
    elif findings:
        for f in findings:
            print(f"{f.file}:{f.line}: [{f.rule}] {f.message}")
        print(f"\nFAILED: {len(findings)} prompt-caching readiness issue(s) detected.")
    else:
        print("PASSED: 0 PCR001-PCR006 violations detected.")

    return 1 if findings else 0


if __name__ == "__main__":
    sys.exit(main())

2. Two-Regime Application Insights KQL Release Gate ​

Evaluates 24-hour AppDependencies OpenTelemetry spans across both pre-GPT-6 (5.6 / 5.4: 0% cached PTU draw, 6x output) and GPT-6 / GPT-6.1 (0.10x / 0.05x cached read, 1.25x write, 5.0x output), reporting:

  • DeployPeakNormTPM & DeployP95NormTPM: Coincident 1-minute normalized TPM across all calls hitting (model_name, sku)—including <1,024-token routers and unkeyed calls—to right-size PTU capacity.
  • HitRatePct, CachedSharePct, ReadWriteRatio, TTFT_p95_ms, ShardPeakNormTPM & GateStatus: Per-node, per-shard cache efficiency against the recommended release SLOs (>=80% hit rate, >=60% cached share, >=10:1 R/W ratio).

Raw KQL download: Download ptu-cache-release-gate.kql

kusto
// Application Insights KQL: Per-node cache efficiency + Two-Regime Shard & Deployment-Wide PTU Normalized TPM
// Last verified: 2026-11-18 21:21 PST | License: MIT
let AllCalls = AppDependencies
| where TimeGenerated > ago(24h) and Type == "AzureOpenAI"
| extend cd = parse_json(Properties)
| extend cache_key = tostring(cd["gen_ai.prompt_cache.key"]), node_name = tostring(cd["agent.node_name"]),
         model_name = tostring(cd["gen_ai.request.model"]), sku = tostring(cd["azure.deployment.sku"]),
         input_tok = tolong(cd["gen_ai.usage.input_tokens"]), output_tok = tolong(cd["gen_ai.usage.output_tokens"]),
         cached_tok = tolong(coalesce(cd["gen_ai.usage.cached_tokens"], 0)),
         write_tok = tolong(coalesce(cd["gen_ai.usage.cache_write_tokens"], 0)),
         ttft_ms = todouble(cd["gen_ai.latency.ttft_ms"])
| extend uncached_tok = max_of(input_tok - cached_tok - write_tok, 0)
| extend norm_tok = case(  // Short-context (<=272K) weights; use 2x input/cache & 7.5x output if input_tok > 272000
         model_name has "6.1", uncached_tok + 0.05 * cached_tok + 1.25 * write_tok + 5.0 * output_tok,
         model_name has "gpt-6", uncached_tok + 0.10 * cached_tok + 1.25 * write_tok + 5.0 * output_tok,
         (input_tok - cached_tok) + 6.0 * output_tok);  // Pre-GPT-6 (5.6 / 5.4): 0x cached, 6x output
let DeployPeak = AllCalls  // Includes <1,024-token & unkeyed calls that share the same PTU deployment
| summarize DeployMinuteTPM = sum(norm_tok) by model_name, sku, bin(TimeGenerated, 1m)
| summarize DeployPeakNormTPM = round(max(DeployMinuteTPM), 0), DeployP95NormTPM = round(percentile(DeployMinuteTPM, 95), 0)
         by model_name, sku;
let CacheCalls = AllCalls | where isnotempty(cache_key) and input_tok >= 1024;
let ShardPeak = CacheCalls
| summarize MinuteNormTPM = sum(norm_tok) by node_name, model_name, sku, cache_key, bin(TimeGenerated, 1m)
| summarize ShardPeakNormTPM = round(max(MinuteNormTPM), 0) by node_name, model_name, sku, cache_key;
CacheCalls
| summarize Calls = count(), HitRatePct = round(100.0 * countif(cached_tok > 0) / count(), 1),
            CachedSharePct = round(100.0 * sum(cached_tok) / max_of(sum(input_tok), 1), 1),
            ReadWriteRatio = iff(sum(write_tok) == 0, real(null), round(todouble(sum(cached_tok)) / sum(write_tok), 1)),
            TTFT_p95_ms = round(percentile(ttft_ms, 95), 0), ShardAvgNormTPM = round(sum(norm_tok) / 1440.0, 0)
         by node_name, model_name, sku, cache_key
| join kind=inner ShardPeak on node_name, model_name, sku, cache_key
| join kind=leftouter DeployPeak on model_name, sku
| extend GateStatus = iff(HitRatePct >= 80.0 and CachedSharePct >= 60.0
                          and (isnull(ReadWriteRatio) or ReadWriteRatio >= 10.0), "PASS", "FAIL")
| project-away node_name1, model_name1, sku1, cache_key1, model_name2, sku2
| order by GateStatus asc, DeployPeakNormTPM desc, ShardPeakNormTPM desc
